cmake_minimum_required(VERSION 3.18)
project(ur7e_nvblox_bridge LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(glog REQUIRED)
execute_process(COMMAND /usr/bin/python3 -c "import pathlib,torch,importlib.util;print(pathlib.Path(torch.__file__).parent);print(pathlib.Path(importlib.util.find_spec('nvblox_torch').origin).parent);print(int(torch.compiled_with_cxx11_abi()))" OUTPUT_VARIABLE paths COMMAND_ERROR_IS_FATAL ANY)
string(STRIP "${paths}" paths)
string(REPLACE "\n" ";" paths "${paths}")
list(GET paths 0 TORCH_ROOT)
list(GET paths 1 WHEEL_ROOT)
list(GET paths 2 TORCH_ABI)
add_library(ur7e_nvblox_bridge SHARED bridge.cpp export.cu)
set_source_files_properties(bridge.cpp PROPERTIES LANGUAGE CUDA)
target_compile_options(ur7e_nvblox_bridge PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--extended-lambda>)
target_compile_definitions(ur7e_nvblox_bridge PRIVATE _GLIBCXX_USE_CXX11_ABI=${TORCH_ABI})
target_include_directories(ur7e_nvblox_bridge PRIVATE ../vendor ../vendor/nvblox/include ../vendor/nvblox_torch/cpp/include "${TORCH_ROOT}/include" "${TORCH_ROOT}/include/torch/csrc/api/include")
target_link_libraries(ur7e_nvblox_bridge PRIVATE Eigen3::Eigen glog::glog CUDA::cudart "${WHEEL_ROOT}/lib/nvblox/libnvblox_lib.so" "${WHEEL_ROOT}/lib/nvblox_torch/cpp/libpy_nvblox.so" "${TORCH_ROOT}/lib/libtorch.so" "${TORCH_ROOT}/lib/libtorch_cpu.so" "${TORCH_ROOT}/lib/libc10.so")
set_target_properties(ur7e_nvblox_bridge PROPERTIES INSTALL_RPATH "${WHEEL_ROOT}/lib/nvblox;${WHEEL_ROOT}/lib/nvblox_torch/cpp;${TORCH_ROOT}/lib")
install(TARGETS ur7e_nvblox_bridge LIBRARY DESTINATION lib)
