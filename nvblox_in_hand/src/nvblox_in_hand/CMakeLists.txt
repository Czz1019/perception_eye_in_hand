cmake_minimum_required(VERSION 3.18)
project(nvblox_in_hand LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(glog REQUIRED)
find_package(gflags REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(nvblox_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
add_library(in_hand_esdf_export STATIC src/esdf_pointcloud_exporter.cu)
target_include_directories(in_hand_esdf_export PUBLIC include)
ament_target_dependencies(in_hand_esdf_export nvblox_ros)
target_link_libraries(in_hand_esdf_export nvblox_ros::nvblox_ros_lib CUDA::cudart)
add_executable(mapper_node src/mapper_node.cpp)
target_include_directories(mapper_node PRIVATE include)
ament_target_dependencies(mapper_node nvblox_ros rclcpp sensor_msgs std_msgs visualization_msgs)
target_link_libraries(mapper_node in_hand_esdf_export nvblox_ros::nvblox_ros_lib CUDA::cudart)
if(BUILD_TESTING)
  add_executable(esdf_pointcloud_export_test ../../tests/esdf_pointcloud_export_test.cu)
  target_link_libraries(esdf_pointcloud_export_test in_hand_esdf_export)
endif()
install(TARGETS mapper_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/robot_pose.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
ament_package()
