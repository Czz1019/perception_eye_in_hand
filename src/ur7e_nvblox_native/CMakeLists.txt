cmake_minimum_required(VERSION 3.18)
project(ur7e_nvblox_native LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(glog REQUIRED)
find_package(gflags REQUIRED)
find_package(rosbag2_cpp REQUIRED)
find_package(nvblox_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(nvblox_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(native_mapper src/native_mapper.cpp src/mesh_view.cpp src/block_export.cu)
target_compile_options(native_mapper PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
ament_target_dependencies(native_mapper nvblox_ros nvblox_msgs rclcpp std_msgs std_srvs sensor_msgs geometry_msgs nav_msgs visualization_msgs)
target_link_libraries(native_mapper nvblox_ros::nvblox_ros_lib CUDA::cudart nlohmann_json::nlohmann_json)
set_target_properties(native_mapper PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
add_executable(mesh_protocol_test src/mesh_protocol_test.cpp)
ament_target_dependencies(mesh_protocol_test nvblox_ros nvblox_msgs visualization_msgs)
install(TARGETS mesh_protocol_test DESTINATION lib/${PROJECT_NAME})
option(UR7E_BUILD_NATIVE_GEOMETRY_TEST "Build standalone CUDA regression executable (does not run it)" OFF)
if(UR7E_BUILD_NATIVE_GEOMETRY_TEST)
  add_executable(native_geometry_test src/native_geometry_test.cpp src/block_export.cu)
  ament_target_dependencies(native_geometry_test nvblox_ros nvblox_msgs rclcpp sensor_msgs visualization_msgs)
  target_compile_options(native_geometry_test PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  target_link_libraries(native_geometry_test nvblox_ros::nvblox_ros_lib CUDA::cudart)
  install(TARGETS native_geometry_test DESTINATION lib/${PROJECT_NAME})
endif()
install(TARGETS native_mapper DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config rviz launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md THIRD_PARTY.md DESTINATION share/${PROJECT_NAME})
ament_package()
